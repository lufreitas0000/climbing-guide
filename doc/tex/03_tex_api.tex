\section{\LaTeX\ Presentation API}

All user-facing layout and database macros are defined via \LaTeX3 (\texttt{expl3}) to guarantee robust argument expansion and prevent fragile token stripping.

\subsection{Macro Specifications}

\begin{sloppypar}
{\small
\begin{itemize}
    \item \textbf{\texttt{\textbackslash CGSetHeaderText\{<header\_text:tl>\}}}
    \begin{itemize}
        \item \textbf{Type}: \texttt{\{tl\}} (Token list string). \textbf{Output}: Mutates global header state.
        \item \textbf{Description}: Overrides running header text displayed in the upper-right margin via \texttt{fancyhdr}.
    \end{itemize}

    \item \textbf{\texttt{\textbackslash CGSidebarTitle\{<title:tl>\}}}
    \begin{itemize}
        \item \textbf{Type}: \texttt{\{tl\}} (Title string). \textbf{Output}: Renders a $4\text{mm} \times 10\text{mm}$ red brand bar with title.
    \end{itemize}

    \item \textbf{\texttt{\textbackslash CGSplitBackground[<ratio:fp>]\{<left\_col:tl>\}\{<right\_col:tl>\}}}
    \begin{itemize}
        \item \textbf{Type}: \texttt{[fp]\{tl\}\{tl\}} (Optional ratio defaulting to \texttt{0.5}, and two colors).
        \item \textbf{Description}: Paints page background in two full-bleed colors using TikZ overlays.
    \end{itemize}

    \item \textbf{\texttt{\textbackslash CGCoverPage\{<image:str>\}\{<title:tl>\}\{<subtitle:tl>\}}}
    \begin{itemize}
        \item \textbf{Type}: \texttt{\{str\}\{tl\}\{tl\}} (Image path, cover title, subtitle).
        \item \textbf{Output}: Full-bleed cover page followed by automatic page break (\texttt{\textbackslash clearpage}).
    \end{itemize}

    \item \textbf{\texttt{\textbackslash CGZoneHeader\{<image:str>\}\{<zone\_name:tl>\}}}
    \begin{itemize}
        \item \textbf{Type}: \texttt{\{str\}\{tl\}} (Banner path, Zone name).
        \item \textbf{Output}: Renders clipped banner, zone title bar, updates Lua state, and writes to ToC (\texttt{\textbackslash addcontentsline\{toc\}\{part\}\{\#2\}}).
    \end{itemize}

    \item \textbf{\texttt{\textbackslash CGSectorHeader\{<color:tl>\}\{<sector\_name:tl>\}}}
    \begin{itemize}
        \item \textbf{Type}: \texttt{\{tl\}\{tl\}} (Color name, Sector name).
        \item \textbf{Output}: Renders western-bordered header, summary blocks, updates Lua state, and writes to ToC (\texttt{\textbackslash addcontentsline\{toc\}\{section\}\{\#2\}}).
    \end{itemize}

    \item \textbf{\texttt{\textbackslash CGRoute\{<name:tl>\}\{<grade:tl>\}\{<len:tl>\}\{<gear:tl>\}\{<setter:tl>\}~\linebreak~
    [<obs:tl>]}}
    \begin{itemize}
        \item \textbf{Type}: \texttt{\{tl\}\{tl\}\{tl\}\{tl\}\{tl\}[tl]} (5 mandatory arguments, 1 optional observation).
        \item \textbf{Output}: Formatted route block with grade indicator badge, auto-scaled title width, and observation box.
    \end{itemize}
\end{itemize}
}
\end{sloppypar}
